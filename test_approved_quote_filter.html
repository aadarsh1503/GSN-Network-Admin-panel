Current Filter Logic

// Filter responses based on quote status
let filteredResponses = Array.isArray(responses) ? responses : [];

// If quote is approved, show only the company that accepted the payment
if (quote && quote.status === 'approved') {
  // First try to find companies with both accepted status and verified payment
  let approvedCompanies = responses.filter(response => 
    response.user_response_status === 'accepted' && 
    response.payment_status === 'verified'
  );
  
  // If no companies found with verified payment, show companies that were accepted
  // (the quote being approved might indicate payment was processed)
  if (approvedCompanies.length === 0) {
    approvedCompanies = responses.filter(response => 
      response.user_response_status === 'accepted'
    );
  }
  
  filteredResponses = approvedCompanies;
}

return filteredResponses; // These companies will be displayed
